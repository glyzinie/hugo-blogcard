{{- $rawURL := .Get 0 | default "" -}} {{- $url := "" -}} {{- $host := "" -}} {{- $card := dict "title" ($rawURL | default "Blog card") "description" "" "image" "" "favicon" "" "unavailable" true -}} {{- with try (urls.Parse $rawURL) -}} {{- if not .Err -}} {{- if and (in (slice "http" "https") (lower .Value.Scheme)) .Value.Host -}} {{- $url = .Value.String -}} {{- $host = .Value.Host -}} {{- $card = partialCached "blogcard/metadata.html" .Value $url -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $url -}} {{- warnf "blogcard: expected an absolute HTTP(S) URL, got %q" $rawURL -}} {{- end -}} {{- $quality := 80 -}} {{- with .Site.Params.imageQuality -}} {{- with try (int .) -}} {{- if not .Err -}} {{- if and (ge .Value 1) (le .Value 100) -}} {{- $quality = .Value -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $thumbnail := partial "blogcard/thumbnail.html" (dict "image" $card.image "fallback" (.Site.Params.defaultNoimage | default "") "quality" $quality) -}} {{- partial "blogcard/styles.html" .Page -}} {{- if $url -}} {{- else -}} {{- end }} {{ $card.title }} {{- with $card.description }} {{ . }} {{- end }} {{- with $host }} {{- with $card.favicon }} {{- end }} {{ $host }} {{- end }} {{- with $thumbnail }} {{- end }} {{- if $url }} {{- else }} {{- end -}}
